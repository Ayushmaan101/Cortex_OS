# ── DB helper ─────────────────────────────────────────────────────────────────
async def adb(method: str, path: str, params=None, json=None, retries=4):
    db = _get_db()
    last_error = None
    for i in range(retries):
        try:
            fn = getattr(db, method)
            r = await fn(path, params=params) if method in ("get", "delete") \
                else await fn(path, params=params, json=json)
            if r.status_code in (200, 201, 204):
                return r
            last_error = f"HTTP {r.status_code}: {r.text[:300]}"
            print(f"  ⚠ DB {method.upper()} {path} → {last_error}")
            break
        except Exception as e:
            last_error = str(e)
            print(f"  ⚠ DB {method.upper()} {path} attempt {i+1}: {e}")
            if i < retries - 1:
                await asyncio.sleep(0.5)
    raise HTTPException(503, f"Database unreachable: {last_error}")


# ── LLM helper ────────────────────────────────────────────────────────────────
def _run_llm_sync(system: str, user: str, max_tokens: int) -> list[str]:
    chunks = []
    stream = llm.chat.completions.create(
        model=LLM_MODEL,
        messages=[{"role": "system", "content": system}, {"role": "user", "content": user}],
        stream=True, max_tokens=max_tokens, temperature=0.4,
    )
    for chunk in stream:
        d = chunk.choices[0].delta.content
        if d:
            chunks.append(d)
    return chunks

async def llm_chunks(system: str, user: str, max_tokens: int = 600) -> list[str]:
    loop = asyncio.get_event_loop()
    return await loop.run_in_executor(_executor, _run_llm_sync, system, user, max_tokens)

def build_branch(nodes: list, root_id: str, depth: int = 0, parent_ids: dict = None) -> str:
    """Build branch with support for multiple parents via edges"""
    node = next((n for n in nodes if n["id"] == root_id), None)
    if not node:
        return ""
    t = f"{'  ' * depth}{'◆' if depth == 0 else '├─'} {node['content']}\n"
    
    # Find children using parent_id OR edges
    children = []
    for n in nodes:
        if n.get("parent_id") == root_id:
            children.append(n)
        elif parent_ids and root_id in parent_ids.get(n["id"], []):
            children.append(n)
    
    for c in children:
        t += build_branch(nodes, c["id"], depth + 1, parent_ids)
    return t

def _bfs_descendants(node_id: str, all_nodes: list, edges: list = None) -> list[str]:
    """BFS considering both parent_id and edges table"""
    children_map: dict[str, list[str]] = {}
    
    # Map from parent_id
    for n in all_nodes:
        pid = n.get("parent_id")
        if pid:
            children_map.setdefault(pid, []).append(n["id"])
    
    # Map from edges table
    if edges:
        for edge in edges:
            children_map.setdefault(edge["source_id"], []).append(edge["target_id"])
    
    to_visit = [node_id]
    result: list[str] = []
    visited = set()
    
    while to_visit:
        cur = to_visit.pop(0)
        if cur in visited:
            continue
        visited.add(cur)
        result.append(cur)
        to_visit.extend(children_map.get(cur, []))
    
    return result

@app.get("/nodes")
async def get_nodes(session_id: str):
    """Fetch nodes with edges information for many-to-one support"""
    r = await adb("get", "/nodes", params={
        "select": "id,content,parent_id,depth,created_at,session_id",
        "session_id": f"eq.{session_id}", "order": "created_at.asc",
    })
    nodes = r.json()
    
    # Fetch edges for this session
    try:
        edges_r = await adb("get", "/edges", params={
            "select": "id,source_id,target_id",
            "session_id": f"eq.{session_id}",
        })
        edges = edges_r.json()
    except:
        edges = []
    
    # Attach parent_ids array to each node
    for node in nodes:
        parent_ids = []
        if node.get("parent_id"):
            parent_ids.append(node["parent_id"])
        # Add parents from edges
        for edge in edges:
            if edge["target_id"] == node["id"] and edge["source_id"] not in parent_ids:
                parent_ids.append(edge["source_id"])
        node["parent_ids"] = parent_ids
    
    return {"nodes": nodes, "edges": edges}

# ── PATCH /nodes/{node_id} — reparent ────────────────────────────────────────
@app.patch("/nodes/{node_id}")
async def patch_node(node_id: str, req: NodePatch):
    session_id = req.session_id
    check = await adb("get", "/nodes", params={"id": f"eq.{node_id}", "session_id": f"eq.{session_id}", "select": "id,depth,parent_id"})
    existing = check.json()
    if not existing:
        raise HTTPException(404, "Node not found in this session")
    old_node = existing[0]
    if req.parent_id is None and req.depth is None:
        return {"ok": True}
    
    # Fetch edges
    try:
        edges_r = await adb("get", "/edges", params={"session_id": f"eq.{session_id}", "select": "id,source_id,target_id"})
        edges = edges_r.json()
    except:
        edges = []
    
    all_r = await adb("get", "/nodes", params={"session_id": f"eq.{session_id}", "select": "id,parent_id,depth"})
    all_nodes = all_r.json()
    id_map = {n["id"]: n for n in all_nodes}
    new_parent = req.parent_id
    if req.depth is not None:
        new_depth = req.depth
    elif new_parent and new_parent in id_map:
        new_depth = id_map[new_parent]["depth"] + 1
    else:
        new_depth = 0
    depth_delta = new_depth - old_node["depth"]
    affected = _bfs_descendants(node_id, all_nodes, edges)
    
    for i, nid in enumerate(affected):
        if nid not in id_map:
            continue
        update: dict = {"depth": id_map[nid]["depth"] + depth_delta}
        if i == 0:
            update["parent_id"] = new_parent
        await adb("patch", "/nodes", params={"id": f"eq.{nid}", "session_id": f"eq.{session_id}"}, json=update)
    
    # Create edge for new parent if not already primary parent
    if new_parent and new_parent != old_node.get("parent_id"):
        try:
            await adb("post", "/edges", json={
                "source_id": new_parent,
                "target_id": node_id,
                "session_id": session_id,
            })
        except:
            pass
    
    return {"ok": True, "affected_nodes": len(affected)}


# ── POST /nodes/{node_id}/detach ─────────────────────────────────────────────
@app.post("/nodes/{node_id}/detach")
async def detach_node(node_id: str, req: NodeDetach):
    session_id = req.session_id
    check = await adb("get", "/nodes", params={"id": f"eq.{node_id}", "session_id": f"eq.{session_id}", "select": "id,depth,parent_id"})
    existing = check.json()
    if not existing:
        raise HTTPException(404, "Node not found")
    old_node = existing[0]
    
    if not old_node.get("parent_id"):
        return {"ok": True, "message": "Node is already a root"}
    
    old_depth = old_node["depth"]
    depth_delta = -old_depth
    
    try:
        edges_r = await adb("get", "/edges", params={"session_id": f"eq.{session_id}", "select": "id,source_id,target_id"})
        edges = edges_r.json()
    except:
        edges = []
    
    all_r = await adb("get", "/nodes", params={"session_id": f"eq.{session_id}", "select": "id,parent_id,depth"})
    all_nodes = all_r.json()
    id_map = {n["id"]: n for n in all_nodes}
    affected = _bfs_descendants(node_id, all_nodes, edges)
    
    for i, nid in enumerate(affected):
        if nid not in id_map:
            continue
        update: dict = {"depth": max(0, id_map[nid]["depth"] + depth_delta)}
        if i == 0:
            update["parent_id"] = None
        await adb("patch", "/nodes", params={"id": f"eq.{nid}", "session_id": f"eq.{session_id}"}, json=update)
    
    # Remove all edges pointing to this node
    try:
        await adb("delete", "/edges", params={"target_id": f"eq.{node_id}", "session_id": f"eq.{session_id}"})
    except:
        pass
    
    return {"ok": True, "affected_nodes": len(affected)}


# ── DELETE /nodes/{node_id} ───────────────────────────────────────────────────
@app.delete("/nodes/{node_id}")
async def delete_node(node_id: str, session_id: str):
    check = await adb("get", "/nodes", params={"id": f"eq.{node_id}", "session_id": f"eq.{session_id}", "select": "id"})
    if not check.json():
        raise HTTPException(404, "Node not found in this session")
    
    try:
        edges_r = await adb("get", "/edges", params={"session_id": f"eq.{session_id}", "select": "id,source_id,target_id"})
        edges = edges_r.json()
    except:
        edges = []
    
    all_r = await adb("get", "/nodes", params={"session_id": f"eq.{session_id}", "select": "id,parent_id"})
    all_nodes = all_r.json()
    
    def count_desc(nid: str) -> int:
        kids = [n for n in all_nodes if n.get("parent_id") == nid]
        for edge in edges:
            if edge["source_id"] == nid:
                target = next((n for n in all_nodes if n["id"] == edge["target_id"]), None)
                if target and target not in kids:
                    kids.append(target)
        return len(kids) + sum(count_desc(k["id"]) for k in kids)
    
    desc_count = count_desc(node_id)
    
    # Delete edges
    try:
        await adb("delete", "/edges", params={"source_id": f"eq.{node_id}", "session_id": f"eq.{session_id}"})
        await adb("delete", "/edges", params={"target_id": f"eq.{node_id}", "session_id": f"eq.{session_id}"})
    except:
        pass
    
    await adb("delete", "/nodes", params={"id": f"eq.{node_id}", "session_id": f"eq.{session_id}"})
    return {"ok": True, "deleted_node_id": node_id, "descendants_deleted": desc_count}


# ── POST /nodes/expand ────────────────────────────────────────────────────────
@app.post("/nodes/expand")
async def expand_node(req: ExpandRequest):
    r = await adb("get", "/nodes", params={"session_id": f"eq.{req.session_id}", "select": "id,content,parent_id,depth", "order": "created_at.asc"})
    all_nodes = r.json()
    target = next((n for n in all_nodes if n["id"] == req.node_id), None)
    if not target:
        raise HTTPException(404, "Node not found")
    children = [n for n in all_nodes if n.get("parent_id") == req.node_id]
    if len(children) > 3:
        raise HTTPException(400, "Node already expanded")
    source_node = next((c for c in children if c["content"].strip().lower().startswith("source:")), None)
    if not source_node:
        raise HTTPException(400, "No 'Source:' child")
    url = source_node["content"].split("Source:", 1)[-1].strip()
    title = target["content"]
    summary_node = next((c for c in children if not c["content"].strip().lower().startswith("source:")), None)
    summary_text = summary_node["content"] if summary_node else "No summary available"
    
    # CONTEXT-AWARE: Include parent context
    parent_context = ""
    if target.get("parent_id"):
        parent = next((n for n in all_nodes if n["id"] == target["parent_id"]), None)
        if parent:
            parent_context = f"\nParent Context: {parent['content']}\n"
    
    prompt = f"""Break down this article into 4-6 distinct key concepts.{parent_context}
Title: {title}\nSummary: {summary_text}\nSource: {url}
Format: CONCEPT: <name> | DETAIL: <1-2 sentence explanation>"""
    
    chunks = await llm_chunks("You are a research analyst extracting key concepts.", prompt, max_tokens=500)
    full_text = "".join(chunks)
    for line in full_text.split("\n"):
        line = line.strip()
        if "CONCEPT:" in line and "DETAIL:" in line:
            try:
                concept = line.split("CONCEPT:")[1].split("|")[0].strip()
                detail = line.split("DETAIL:")[1].strip()
                nc = f"{concept}: {detail}"
                await adb("post", "/nodes", json={"content": nc, "parent_id": req.node_id, "depth": target["depth"] + 1, "session_id": req.session_id, "embedding": get_embedding(nc)})
            except Exception as e:
                print(f"  ⚠ {e}")
    def stream():
        for c in chunks:
            yield c
    return StreamingResponse(stream(), media_type="text/plain")

# ── POST /synthesize ──────────────────────────────────────────────────────────
@app.post("/synthesize")
async def synthesize(req: SynthesizeRequest):
    r = await adb("get", "/nodes", params={"select": "id,content,parent_id,depth", "session_id": f"eq.{req.session_id}", "order": "created_at.asc"})
    all_nodes = r.json()
    
    # CONTEXT-AWARE: Get parent chain context
    target_node = next((n for n in all_nodes if n["id"] == req.node_id), None)
    parent_context = ""
    if target_node and target_node.get("parent_id"):
        parent_chain = []
        current_id = target_node["parent_id"]
        depth = 0
        while current_id and depth < 3:  # Max 3 levels up
            parent = next((n for n in all_nodes if n["id"] == current_id), None)
            if parent:
                parent_chain.append(parent["content"])
                current_id = parent.get("parent_id")
            else:
                break
            depth += 1
        if parent_chain:
            parent_context = "Context from parent nodes:\n" + "\n".join(f"  ↑ {pc}" for pc in reversed(parent_chain)) + "\n\n"
    
    try:
        edges_r = await adb("get", "/edges", params={"session_id": f"eq.{req.session_id}", "select": "source_id,target_id"})
        edges = edges_r.json()
        parent_ids = {}
        for edge in edges:
            parent_ids.setdefault(edge["target_id"], []).append(edge["source_id"])
    except:
        parent_ids = {}
    
    branch = build_branch(all_nodes, req.node_id, parent_ids=parent_ids)
    if not branch.strip():
        raise HTTPException(404, "Node not found")
    
    prompt = f"{parent_context}Synthesize:\n\n{branch}"
    chunks = await llm_chunks(TONE_PROMPTS[req.tone], prompt)
    
    def stream():
        for c in chunks:
            yield c
    return StreamingResponse(stream(), media_type="text/plain")

@app.post("/nodes/arrange")
async def arrange_nodes(req: ArrangeRequest):
    r = await adb("get", "/nodes", params={"session_id": f"eq.{req.session_id}", "select": "id,content,parent_id,depth", "order": "created_at.asc"})
    all_nodes = r.json()
    if len(all_nodes) < 3:
        return {"suggestions": [], "message": "Need at least 3 nodes to suggest arrangement"}
    
    node_list = "\n".join(
        f"ID:{n['id'][:8]} DEPTH:{n['depth']} PARENT:{(n['parent_id'] or 'ROOT')[:8]} CONTENT:{n['content'][:80]}"
        for n in all_nodes[:40]
    )
    if req.mode == "coarse":
        system = "You are a knowledge graph organizer. Group nodes into broad topic clusters. Focus on thematic similarity."
        prompt = f"""Analyze these knowledge nodes and suggest groupings by broad topic.
Return ONLY a JSON array: [{{"source_id":"abc12345","target_parent_id":"def67890","reason":"shares theme: [topic]"}}]
Use 'ROOT' for target_parent_id if node should be a root. Max 10 suggestions.
Prioritize: semantic clustering, topic coherence, reducing redundancy.
Nodes:\n{node_list}"""
    elif req.mode == "fine":
        system = "You are a knowledge graph organizer. Find granular subtopic relationships and hierarchical dependencies."
        prompt = f"""Suggest rearrangements to create detailed subtopic hierarchies.
Return ONLY JSON array: [{{"source_id":"abc12345","target_parent_id":"def67890","reason":"is subtopic/component of: [concept]"}}]
Max 10 suggestions. Focus on: part-of relationships, prerequisite chains, conceptual dependencies.
Nodes:\n{node_list}"""
    else:
        system = "You are a knowledge graph expert finding hidden conceptual bridges and cross-domain connections."
        prompt = f"""Find non-obvious cross-connections between nodes that share concepts, methodologies, or implications.
Return ONLY JSON array: [{{"source_id":"abc12345","target_parent_id":"def67890","reason":"connected via: [bridge concept/method]"}}]
Max 8 suggestions. Focus on: analogies, shared frameworks, causal chains, complementary ideas.
Avoid obvious parent-child relationships.
Nodes:\n{node_list}"""
    
    chunks = await llm_chunks(system, prompt, max_tokens=700)
    full_text = "".join(chunks)
    suggestions = []
    try:
        match = re.search(r'\[.*\]', full_text, re.DOTALL)
        if match:
            raw = json.loads(match.group())
            id_prefix_map = {n["id"][:8]: n["id"] for n in all_nodes}
            for s in raw:
                src = id_prefix_map.get(s.get("source_id", ""))
                tgt_raw = s.get("target_parent_id", "ROOT")
                tgt = None if tgt_raw == "ROOT" else id_prefix_map.get(tgt_raw)
                if src:
                    suggestions.append({
                        "source_id": src,
                        "target_parent_id": tgt,
                        "reason": s.get("reason", ""),
                        "source_content": next((n["content"][:60] for n in all_nodes if n["id"] == src), ""),
                        "target_content": next((n["content"][:60] for n in all_nodes if n["id"] == tgt), "ROOT") if tgt else "ROOT",
                    })
    except Exception as e:
        print(f"  ⚠ arrange parse error: {e}")
    return {"suggestions": suggestions, "mode": req.mode}

@app.post("/harvest/run")
async def trigger_harvest(req: HarvestRequest, background_tasks: BackgroundTasks):
    if _harvest_status["running"]:
        raise HTTPException(409, "Harvest already running")
    async def _run():
        from harvester import run_harvest
        _harvest_status["running"] = True
        _harvest_status["last_run"] = datetime.now(timezone.utc).isoformat()
        try:
            items = await run_harvest(req.source_ids)
            existing = {i["id"] for i in _inbox}
            _inbox[:0] = [i for i in items if i["id"] not in existing]
            _harvest_status["item_count"] = len(_inbox)
        except Exception as e:
            print(f"Harvest error: {e}")
        finally:
            _harvest_status["running"] = False
    background_tasks.add_task(_run)
    return {"started": True}

@app.post("/harvest/inject")
async def inject_article(req: InjectRequest):
    article = next((i for i in _inbox if i["id"] == req.article_id), None)
    if not article:
        raise HTTPException(404, "Article not found in inbox")
    root_content = article["title"]
    r = await adb("post", "/nodes", json={"content": root_content, "parent_id": None, "depth": 0, "session_id": req.session_id, "embedding": get_embedding(root_content)})
    data = r.json()
    root_id = (data[0] if isinstance(data, list) else data)["id"]
    url_node = f"Source: {article['url']}"
    await adb("post", "/nodes", json={"content": url_node, "parent_id": root_id, "depth": 1, "session_id": req.session_id, "embedding": get_embedding(url_node)})
    if article.get("summary"):
        await adb("post", "/nodes", json={"content": article["summary"], "parent_id": root_id, "depth": 1, "session_id": req.session_id, "embedding": get_embedding(article["summary"])})
    if req.depth == "deep":
        prompt = f"""Break down into 4-6 key concepts.
Title: {article['title']}\nSummary: {article.get('summary', 'N/A')}\nSource: {article['url']}
Format: CONCEPT: <name> | DETAIL: <explanation>"""
        chunks = await llm_chunks("You are a research analyst.", prompt, max_tokens=500)
        full_text = "".join(chunks)
        for line in full_text.split("\n"):
            if "CONCEPT:" in line and "DETAIL:" in line:
                try:
                    concept = line.split("CONCEPT:")[1].split("|")[0].strip()
                    detail = line.split("DETAIL:")[1].strip()
                    nc = f"{concept}: {detail}"
                    await adb("post", "/nodes", json={"content": nc, "parent_id": root_id, "depth": 1, "session_id": req.session_id, "embedding": get_embedding(nc)})
                except Exception:
                    pass
        _inbox[:] = [i for i in _inbox if i["id"] != req.article_id]
        def stream_analysis():
            for c in chunks:
                yield c
        return StreamingResponse(stream_analysis(), media_type="text/plain")
    _inbox[:] = [i for i in _inbox if i["id"] != req.article_id]
    return {"ok": True, "root_id": root_id}

async def _scheduled_harvest():
    await asyncio.sleep(90)
    while True:
        _harvest_status["next_run"] = (datetime.now(timezone.utc) + timedelta(hours=6)).isoformat()
        if not _harvest_status["running"]:
            _harvest_status["running"] = True
            try:
                from harvester import run_harvest
                items = await run_harvest()
                existing = {i["id"] for i in _inbox}
                _inbox[:0] = [i for i in items if i["id"] not in existing]
                _harvest_status["last_run"] = datetime.now(timezone.utc).isoformat()
                _harvest_status["item_count"] = len(_inbox)
            except Exception as e:
                print(f"Scheduled harvest error: {e}")
            finally:
                _harvest_status["running"] = False
        await asyncio.sleep(6 * 3600)
